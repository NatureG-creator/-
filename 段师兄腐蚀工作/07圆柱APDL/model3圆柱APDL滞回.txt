        c = nx-1

! OUTRES,ALL,LAST

! 5. 后处理 POST26 中的转角—弯矩曲线。
!
! 相比 OUTRES,ALL,ALL：
! 不保存每个自动子步的全场结果，文件小很多。
! ===========================================================

! ===========================================================
! VARIABLE AMPLITUDE CYCLIC ROTATION PARAMETERS
! 变幅滞回加载：0.01 -> 0.15 rad
!
! branch_div = 3：
! 比 12 快很多；
! 又比 1 的端点连接更平滑。
!
! 加载路径：
! 0 -> +amp -> -amp -> +amp
! 最后从最终 +amp 回到 0
! ===========================================================

amp_start = 0.02
amp_end   = 0.20
amp_inc   = 0.02

cycles_per_amp = 1
branch_div = 2



! 幅值级数
namp = (amp_end - amp_start)/amp_inc + 1

! 每个幅值每个循环包含 3 个分支：
! current -> +amp
! +amp -> -amp
! -amp -> +amp
! 最后再从最终 +amp -> 0
nstate = 1 + namp*cycles_per_amp*3*branch_div + branch_div

*DIM,rot_vals,ARRAY,nstate

idx_load = 1
rot_vals(idx_load) = 0
rot_prev = 0

*DO,ia,1,namp

    amp_now = amp_start + (ia-1)*amp_inc

    *DO,icyc,1,cycles_per_amp

        ! Branch 1: current rotation -> +amp_now
        rot_target = amp_now
        *DO,j,1,branch_div
            idx_load = idx_load + 1
            rot_vals(idx_load) = rot_prev + (rot_target-rot_prev)*j/branch_div
        *ENDDO
        rot_prev = rot_target

        ! Branch 2: +amp_now -> -amp_now
        rot_target = -amp_now
        *DO,j,1,branch_div
            idx_load = idx_load + 1
            rot_vals(idx_load) = rot_prev + (rot_target-rot_prev)*j/branch_div
        *ENDDO
        rot_prev = rot_target

        ! Branch 3: -amp_now -> +amp_now
        rot_target = amp_now
        *DO,j,1,branch_div
            idx_load = idx_load + 1
            rot_vals(idx_load) = rot_prev + (rot_target-rot_prev)*j/branch_div
        *ENDDO
        rot_prev = rot_target

    *ENDDO

*ENDDO

! 最终从最后一级正向幅值回到 0
rot_target = 0
*DO,j,1,branch_div
    idx_load = idx_load + 1
    rot_vals(idx_load) = rot_prev + (rot_target-rot_prev)*j/branch_div
*ENDDO

nstate = idx_load

! ===========================================================

*CFOPEN,progress_hysteresis,txt
*VWRITE,'Step','TotalStep','Amplitude','Rotation','Status'
(A8,',',A10,',',A12,',',A12,',',A12)

    /COM, ==================================================
    *STATUS,i
    *STATUS,nstate
    *STATUS,rot_now

    SOLVE

    *CFOPEN,progress_hysteresis,txt,,APPEND
    *VWRITE,i,nstate,amp_abs,rot_now,'DONE'
    (I8,',',I10,',',E12.5,',',E12.5,',',A12)
    *CFCLOS

*ENDDO
